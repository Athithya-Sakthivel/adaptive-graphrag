# Block 2 — self-contained GPU block: clone, download, chunk, embed, upload
# Accelerator: GPU T4 x2 (Settings → Accelerator → Save; session restarts)
# Works even if /kaggle/working was wiped on restart.
#
# Uses --device cuda:0 (single GPU). For 200-2,400 texts, single-GPU with
# FP16 is faster than the multi-GPU pool: the pool's ~20 s startup exceeds
# any parallel gain, and the pool's CUDA-context interaction with
# empty_cache() can deadlock on tail batches below MULTI_GPU_MIN_TEXTS.
#
# Verifies ALL five required fixes on origin/main before running:
#   1. pipeline_helpers.py: parenthesized except clause
#   2. pipeline_helpers.py: worker_init accepts gpu_model_dir_str
#   3. pipeline.py: initargs passes both model_dir and gpu_model_dir
#   4. embedder.py: MULTI_GPU_MIN_TEXTS = 20_000
#   5. embedder.py: empty_cache() guarded behind self._pool is None
# ════════════════════════════════════════════════════════════════════

# ── 2. Verify required fixes ────────────────────────────────────────
import re
import ast as _ast

helpers_src  = open(HELPERS).read()
pipeline_src = open(PIPELINE).read()
embedder_src = open(EMBEDDER).read()

missing: list[str] = []

# 1. Parenthesized except clause
if "except (TypeError, ValueError, OverflowError):" not in helpers_src:
    missing.append("pipeline_helpers: parenthesized except")

# 2. worker_init accepts gpu_model_dir_str
if not re.search(r"def\s+worker_init\s*\([^)]*gpu_model_dir_str", helpers_src, re.DOTALL):
    missing.append("pipeline_helpers: worker_init missing gpu_model_dir_str")

# 3. initargs passes both model dirs (AST-based)
try:
    _tree = _ast.parse(pipeline_src)
except SyntaxError as _e:
    missing.append(f"pipeline: syntax error: {_e}")
else:
    _has_initargs = False
    for _n in _ast.walk(_tree):
        if isinstance(_n, _ast.Call) and getattr(_n.func, "attr", "") == "ProcessPoolExecutor":
            for _kw in _n.keywords:
                if _kw.arg == "initargs" and isinstance(_kw.value, _ast.Tuple):
                    _segs = [
                        _ast.get_source_segment(pipeline_src, e) or ""
                        for e in _kw.value.elts
                    ]
                    if (any("model_dir" in s for s in _segs)
                        and any("gpu_model_dir" in s for s in _segs)):
                        _has_initargs = True
    if not _has_initargs:
        missing.append("pipeline: initargs missing model_dir/gpu_model_dir")

# 4. MULTI_GPU_MIN_TEXTS threshold
if "MULTI_GPU_MIN_TEXTS = 20_000" not in embedder_src:
    missing.append("embedder: MULTI_GPU_MIN_TEXTS = 20_000")

# 5. empty_cache guard
if 'self.device.startswith("cuda") and self._pool is None' not in embedder_src:
    missing.append("embedder: empty_cache not guarded behind pool-is-None")

if missing:
    for m in missing:
        print(f"MISSING FIX on origin/main: {m}")
    raise SystemExit("origin/main is missing required fixes — push from laptop and re-run")

# Final syntax validation of all three files
_ast.parse(helpers_src)
_ast.parse(pipeline_src)
_ast.parse(embedder_src)

print("fixes present: except, worker_init, initargs, MULTI_GPU_MIN_TEXTS=20k, empty_cache guard")
print(f"embed device: {EMBED_DEVICE} (single GPU)")

# Verify chunked/ complete and every child under 512 real tokens
import json
import statistics
from pathlib import Path
from transformers import AutoTokenizer

chunked_dir = f"{DATA}/chunked"
assert os.path.isdir(chunked_dir), f"chunked dir missing: {chunked_dir}"

case_files = [f for f in os.listdir(chunked_dir)
              if f.endswith(".jsonl") and not f.startswith("_")]
print(f"\nchunked case files: {len(case_files)}")
assert len(case_files) > 0, "no chunked files produced"
assert os.path.isfile(f"{chunked_dir}/_manifest.jsonl"), "chunked/_manifest.jsonl missing"
assert not os.path.isfile(f"{chunked_dir}/_HALTED"), "chunk halted — check _failures.jsonl"

fail_path = f"{chunked_dir}/_failures.jsonl"
if os.path.isfile(fail_path):
    with open(fail_path) as f:
        n = sum(1 for _ in f)
    assert n == 0, f"{n} chunk failures — resolve before embed"

tok = AutoTokenizer.from_pretrained(GPU, local_files_only=True)
tokens: list[int] = []
over:  list[tuple[str, int]] = []
for p in sorted(Path(chunked_dir).glob("*.jsonl")):
    if p.name.startswith("_"):
        continue
    for line in p.read_text().splitlines():
        row = json.loads(line)
        if row["is_parent"]:
            continue
        n = len(tok.encode(row["text"], add_special_tokens=True, truncation=False))
        tokens.append(n)
        if n > 512:
            over.append((row["chunk_id"], n))

print(f"children: {len(tokens)}  p50={statistics.median(tokens):.0f}  "
      f"p95={sorted(tokens)[int(.95*len(tokens))]:.0f}  max={max(tokens)}")
assert len(over) == 0, f"{len(over)} chunks exceed 512 tokens — do not embed"

# ── 8. Run embed on GPU (single device) ─────────────────────────────

fail_path = f"{embedded_dir}/_failures.jsonl"
if os.path.isfile(fail_path):
    with open(fail_path) as f:
        n = sum(1 for _ in f)
    assert n == 0, f"{n} embed failures"

# Sample vector sanity: 1024 dims, L2-normalized
with open(f"{embedded_dir}/{embedded_files[0]}") as f:
    vec = json.loads(f.readline())["vector"]
assert len(vec) == 1024, f"expected 1024 dims, got {len(vec)}"
norm = sum(x * x for x in vec) ** 0.5
assert abs(norm - 1.0) < 1e-3, f"vector not L2-normalized: norm={norm}"
print(f"sample vector: dims={len(vec)}  L2 norm={norm:.4f}")

# ── 10. Upload chunked/ + embedded/ as tarballs ─────────────────────
#
# Per-file upload hits HF's per-request rate limit (~1000 req/5min per IP).
# 1553 cases × 2 files + metadata = ~3100 files × 2 requests each (GET+HEAD)
# = ~6200 requests → unavoidable 429 storm.
#
# Tarball approach: 2 tarballs, 2 uploads, ~2 requests total. The laptop
# extracts them and gets the exact same per-case JSONL layout.
#
# We also delete any stale per-file artifacts from previous runs, so the
# repo contains only the tarballs going forward.

print(f"\nuploading to {REPO_ARTIFACTS} ...")

# Ensure repo exists (use the non-deprecated `hf repos create`)
subprocess.run(
    ["hf", "repos", "create", REPO_ARTIFACTS,
     "--repo-type", "dataset", "--private", "--exist-ok"],
    check=False, env=os.environ,
)

# Package as tarballs
print("packaging artifacts as tarballs...")
tarballs = [
    ("chunked.tar.gz", "chunked"),
    ("embedded.tar.gz", "embedded"),
]
for tarball_name, subdir in tarballs:
    tarball_path = f"{DATA}/{tarball_name}"
    run(["tar", "-czf", tarball_path, "-C", DATA, subdir])
    size_mb = os.path.getsize(tarball_path) / 1e6
    print(f"  {tarball_name}: {size_mb:.1f} MB")

# Upload both tarballs (2 requests)
for tarball_name, _ in tarballs:
    run(["hf", "upload", REPO_ARTIFACTS,
         f"{DATA}/{tarball_name}", tarball_name,
         "--repo-type", "dataset"])

# Verify upload
from huggingface_hub import HfApi

api = HfApi(token=os.environ["HF_TOKEN"])
remote_files = list(api.list_repo_files(REPO_ARTIFACTS, repo_type="dataset"))

for tarball_name, _ in tarballs:
    assert tarball_name in remote_files, f"{tarball_name} not uploaded"
print(f"\nremote tarballs: {sorted(f for f in remote_files if f.endswith('.tar.gz'))}")

# Clean up stale per-file artifacts from prior runs. `delete_patterns`
# batches this into a single commit (1 request) regardless of file count.
stale_files = [
    f for f in remote_files
    if (f.startswith("chunked/") or f.startswith("embedded/")) and f.endswith(".jsonl")
]
if stale_files:
    print(f"deleting {len(stale_files)} stale per-file artifacts from previous runs...")
    try:
        api.delete_files(
            repo_id=REPO_ARTIFACTS,
            repo_type="dataset",
            delete_patterns=["chunked/*.jsonl", "embedded/*.jsonl"],
            commit_message="remove per-file artifacts (superseded by tarballs)",
        )
        print("  stale files deleted")
    except Exception as exc:
        print(f"  WARNING: could not delete stale files: {type(exc).__name__}: {exc}")
        print("  (harmless — laptop download script only fetches tarballs)")

# Summary
print(f"\nlocal chunked: {len(case_files)}  embedded: {len(embedded_files)}")
for tarball_name, _ in tarballs:
    size_mb = os.path.getsize(f"{DATA}/{tarball_name}") / 1e6
    print(f"  uploaded {tarball_name}: {size_mb:.1f} MB")

# Clean up local tarballs (they're on HF now; laptop re-downloads)
for tarball_name, _ in tarballs:
    try:
        os.remove(f"{DATA}/{tarball_name}")
    except FileNotFoundError:
        pass

print()
print("✓ Block 2 complete.")
print(f"https://huggingface.co/datasets/{REPO_ARTIFACTS}")
