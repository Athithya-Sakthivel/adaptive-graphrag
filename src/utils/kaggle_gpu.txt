# Uses --device cuda:0 (single GPU). For 200-2,400 texts, single-GPU with
# FP16 is faster than the multi-GPU pool: the pool's ~20 s startup exceeds
# any parallel gain, and the pool's CUDA-context interaction with
# empty_cache() can deadlock on tail batches below MULTI_GPU_MIN_TEXTS.
#
# Verifies ALL five required fixes on origin/main before running:
#   1. pipeline_helpers.py: parenthesized except clause
#   2. pipeline_helpers.py: worker_init accepts gpu_model_dir_str
#   3. pipeline.py: initargs passes both model_dir and gpu_model_dir
#   4. embedder.py: MULTI_GPU_MIN_TEXTS = 20_000
#   5. embedder.py: empty_cache() guarded behind self._pool is None
# ════════════════════════════════════════════════════════════════════

# Verify chunked/ complete and every child under 512 real tokens
import json
import statistics
from pathlib import Path
from transformers import AutoTokenizer

chunked_dir = f"{DATA}/chunked"
assert os.path.isdir(chunked_dir), f"chunked dir missing: {chunked_dir}"

case_files = [f for f in os.listdir(chunked_dir)
              if f.endswith(".jsonl") and not f.startswith("_")]
print(f"\nchunked case files: {len(case_files)}")
assert len(case_files) > 0, "no chunked files produced"
assert os.path.isfile(f"{chunked_dir}/_manifest.jsonl"), "chunked/_manifest.jsonl missing"
assert not os.path.isfile(f"{chunked_dir}/_HALTED"), "chunk halted — check _failures.jsonl"

fail_path = f"{chunked_dir}/_failures.jsonl"
if os.path.isfile(fail_path):
    with open(fail_path) as f:
        n = sum(1 for _ in f)
    assert n == 0, f"{n} chunk failures — resolve before embed"

tok = AutoTokenizer.from_pretrained(GPU, local_files_only=True)
tokens: list[int] = []
over:  list[tuple[str, int]] = []
for p in sorted(Path(chunked_dir).glob("*.jsonl")):
    if p.name.startswith("_"):
        continue
    for line in p.read_text().splitlines():
        row = json.loads(line)
        if row["is_parent"]:
            continue
        n = len(tok.encode(row["text"], add_special_tokens=True, truncation=False))
        tokens.append(n)
        if n > 512:
            over.append((row["chunk_id"], n))

print(f"children: {len(tokens)}  p50={statistics.median(tokens):.0f}  "
      f"p95={sorted(tokens)[int(.95*len(tokens))]:.0f}  max={max(tokens)}")
assert len(over) == 0, f"{len(over)} chunks exceed 512 tokens — do not embed"

    vec = json.loads(f.readline())["vector"]
