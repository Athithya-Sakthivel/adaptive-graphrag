# Verify chunked/ complete and token sizes
import json, statistics
from pathlib import Path
from transformers import AutoTokenizer

chunked_dir = f"{DATA}/chunked"
assert os.path.isdir(chunked_dir)
case_files = [f for f in os.listdir(chunked_dir)
              if f.endswith(".jsonl") and not f.startswith("_")]
print(f"\nchunked case files: {len(case_files)}")
assert len(case_files) > 0
assert os.path.isfile(f"{chunked_dir}/_manifest.jsonl")
assert not os.path.isfile(f"{chunked_dir}/_HALTED")

fail_path = f"{chunked_dir}/_failures.jsonl"
if os.path.isfile(fail_path):
    with open(fail_path) as f:
        n = sum(1 for _ in f)
    assert n == 0, f"{n} chunk failures — resolve before embed"

tok = AutoTokenizer.from_pretrained(GPU, local_files_only=True)
tokens, over = [], []
for p in sorted(Path(chunked_dir).glob("*.jsonl")):
    if p.name.startswith("_"):
        continue
    for line in p.read_text().splitlines():
        row = json.loads(line)
        if row["is_parent"]:
            continue
        n = len(tok.encode(row["text"], add_special_tokens=True, truncation=False))
        tokens.append(n)
        if n > 512:
            over.append((row["chunk_id"], n))

print(f"children: {len(tokens)}  p50={statistics.median(tokens):.0f}  "
      f"p95={sorted(tokens)[int(.95*len(tokens))]:.0f}  max={max(tokens)}")
assert len(over) == 0, f"{len(over)} chunks exceed 512 tokens — do not embed"

# Sample vector sanity

assert len(vec) == 1024
norm = sum(x*x for x in vec) ** 0.5
assert abs(norm - 1.0) < 1e-3
